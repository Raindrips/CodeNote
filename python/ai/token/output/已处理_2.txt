## 二、核心功能实现

### 1. 背包系统（InventoryManager.gd）
```gd
extends Node

class_name InventoryManager

var items = {} # {item_id: {count: int, item: Item}}
var max_slots = 20
var item_db = {}

func _ready():
    _load_item_database()
    _initialize_inventory()

func _load_item_database():
    var file = File.new()
    if file.file_exists("res://assets/item_database.json"):
        file.open("res://assets/item_database.json", File.READ)
        item_db = JSON.parse(file.get_as_text()).result
        file.close()

func _initialize_inventory():
    for i in range(max_slots):
        items[i] = {count: 0, item: null}

func add_item(item_id: String, count: int = 1) -> bool:
    for slot in items:
        if items[slot].item == null or items[slot].item.id == item_id:
            if items[slot].item == null:
                items[slot].item = item_db[item_id]
                items[slot].count = count
            else:
                items[slot].count += count
            _update_ui()
            return true
    return false

func remove_item(item_id: String, count: int = 1) -> bool:
    for slot in items:
        if items[slot].item and items[slot].item.id == item_id:
            if items[slot].count > count:
                items[slot].count -= count
            else:
                items[slot].item = null
                items[slot].count = 0
            _update_ui()
            return true
    return false

func get_item_count(item_id: String) -> int:
    for slot in items:
        if items[slot].item and items[slot].item.id == item_id:
            return items[slot].count
    return 0

func _update_ui():
    # 触发UI更新信号
    emit_signal("inventory_updated")
```

**优化技巧：**
- 使用字典替代数组提高查找效率
- 添加物品类型分类（如消耗品/装备/材料）
- 实现物品堆叠上限限制
- 使用对象池管理UI槽位

**常见坑点：**
- 避免直接修改items字典导致UI不更新
- 注意JSON数据的类型转换问题
- 需要处理物品ID不存在的情况

### 2. 合成系统（CraftingSystem.gd）
```gd
extends Node

class_name CraftingSystem

var recipes = {} # {recipe_id: {inputs: Dictionary, output: String, count: int}}
var crafting_area = null

func _ready():
    _load_recipes()
    crafting_area = get_node("CraftingArea")

func _load_recipes():
    var file = File.new()
    if file.file_exists("res://assets/recipes.json"):
        file.open("res://assets/recipes.json", File.READ)
        recipes = JSON.parse(file.get_as_text()).result
        file.close()

func craft_recipe(recipe_id: String) -> bool:
    var recipe = recipes[recipe_id]
    if _check_requirements(recipe.inputs):
        for item_id in recipe.inputs:
            remove_item(item_id, recipe.inputs[item_id])
        add_item(recipe.output, recipe.count)
        return true
    return false

func _check_requirements(requirements: Dictionary) -> bool:
    for item_id in requirements:
        if get_item_count(item_id) < requirements[item_id]:
            return false
    return true

func on_crafting_complete():
    # 触发合成完成事件
    emit_signal("recipe_crafted")
```

### 3. 商店系统（ShopSystem.gd）
```gd
extends Node

class_name ShopSystem

var shop_items = {} # {item_id: {price: int, stock: int}}
var currency = 1000

func _ready():
    _load_shop_data()

func _load_shop_data():
    var file = File.new()
    if file.file_exists("res://assets/shop_database.json"):
        file.open("res://assets/shop_database.json", File.READ)
        shop_items = JSON.parse(file.get_as_text()).result
        file.close()

func buy_item(item_id: String, amount: int = 1) -> bool:
    if shop_items.has(item_id) and shop_items[item_id].stock >= amount:
        if currency >= shop_items[item_id].price * amount:
            currency -= shop_items[item_id].price * amount
            add_item(item_id, amount)
            shop_items[item_id].stock -= amount
            _update_ui()
            return true
    return false

func sell_item(item_id: String, amount: int = 1) -> bool:
    if get_item_count(item_id) >= amount:
        currency += shop_items[item_id].price * amount
        remove_item(item_id, amount)
        shop_items[item_id].stock += amount
        _update_ui()
        return true
    return false
```

## 三、数据持久化系统（SaveSystem.gd）
```gd
extends Node

class_name SaveSystem

var save_file = "res://savegame.json"
var encryption_key = "supersecretkey123456789"

func save_game(inventory: Dictionary, shop: Dictionary, currency: int):
    var data = {
        "inventory": inventory,
        "shop": shop,
        "currency": currency
    }
    
    # 加密处理
    var encrypted = _encrypt(JSON.stringify(data))
    
    # 保存到文件
    var file = File.new()
    if file.file_exists(save_file):
        file.remove(save_file)
    
    file.open(save_file, File.WRITE)
    file.store_string(encrypted)
    file.close()

func load_game() -> Dictionary:
    var file = File.new()
    if file.file_exists(save_file):
        file.open(save_file, File.READ)
        var encrypted = file.get_as_text()
        file.close()
        
        var data = _decrypt(encrypted)
        return JSON.parse(data).result
    return {}

func _encrypt(text: String) -> String:
    # 实际开发中应使用更安全的加密算法
    return Base64.encode_text(text + encryption_key)

func _decrypt(text: String) -> String:
    var decoded = Base64.decode_text(text)
    return decoded.replace(encryption_key, "")
```

**性能优化：**
- 使用二进制格式替代JSON提升读写速度
- 添加增量保存机制（仅保存变化数据）
- 实现版本兼容性校验
- 使用异步IO处理大文件

## 五、网络同步方案（NetworkSync.gd）
```gd
extends Node

class_name NetworkSync

var ws = WebSocket.new()

func _ready():
    ws.connect("connected", self, "_on_ws_connected")
    ws.connect("text_received", self, "_on_text_received")
    ws.connect("disconnected", self, "_on_ws_disconnected")
    ws.connect("error", self, "_on_ws_error")

func _on_ws_connected():
    print("Connected to server")
    _send_initial_data()

func _send_initial_data():
    var data = {
        "inventory": InventoryManager.instance.items,
        "shop": ShopSystem.instance.shop_items,
        "currency": ShopSystem.instance.currency
    }
    ws.send_text(JSON.stringify(data))

func _on_text_received(data: String):
    var received_data = JSON.parse(data).result
    # 合并数据到本地
    _merge_data(received_data)
```

**同步策略建议：**
- 使用Delta同步优化数据传输
- 实现冲突解决机制（时间戳/版本号）
- 添加网络状态检测
- 使用Protobuf替代JSON提升传输效率
