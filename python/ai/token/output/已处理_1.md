### 3.2 核心控制脚本
```csharp
using UnityEngine;

public class PlayerController : MonoBehaviour
{
    [Header("Movement")]
    public float moveSpeed = 5f;
    public float jumpForce = 10f;
    public float wallJumpForce = 8f;
    
    [Header("Physics")]
    public LayerMask groundLayer;
    public LayerMask wallLayer;
    
    [Header("Animation")]
    public Animator animator;
    
    private Rigidbody2D rb;
    private bool isGrounded;
    private bool isWallJumping;
    private int jumpCount = 0;
    private float horizontalInput;
    private Vector2 wallJumpDirection;
    
    void Start()
    {
        rb = GetComponent<Rigidbody2D>();
        animator = GetComponent<Animator>();
    }

    void Update()
    {
        horizontalInput = Input.GetAxis("Horizontal");
        
        // 双跳检测
        if (Input.GetButtonDown("Jump") && jumpCount < 2)
        {
            rb.velocity = new Vector2(rb.velocity.x, jumpForce);
            jumpCount++;
            animator.SetBool("IsJumping", true);
        }
        
        // 墙跳检测
        if (Input.GetButtonDown("Jump") && isWallJumping)
        {
            rb.velocity = new Vector2(wallJumpDirection.x * wallJumpForce, wallJumpForce);
            isWallJumping = false;
            jumpCount = 1; // 重置跳跃次数
            animator.SetBool("IsJumping", true);
        }
    }

    void FixedUpdate()
    {
        // 水平移动
        rb.velocity = new Vector2(horizontalInput * moveSpeed, rb.velocity.y);
        
        // 碰撞检测
        isGrounded = Physics2D.OverlapCircle(transform.position, 0.1f, groundLayer);
        isWallJumping = Physics2D.OverlapCircle(transform.position, 0.1f, wallLayer);
        
        // 动画控制
        animator.SetFloat("Speed", Mathf.Abs(horizontalInput));
        animator.SetBool("IsGrounded", isGrounded);
    }
}
```

### 3.3 性能优化技巧
- 使用`Physics2D.OverlapCircle`替代多次射线检测
- 在FixedUpdate中处理物理相关逻辑
- 使用对象池管理频繁创建的物体
- 为碰撞体设置合适的LayerMask

## 六、相机跟随系统
```csharp
using UnityEngine;

public class CameraFollow : MonoBehaviour
{
    public Transform player;
    public Vector3 offset = new Vector3(0, 2, -10);
    public float smoothSpeed = 0.125f;
    
    void LateUpdate()
    {
        Vector3 desiredPosition = player.position + offset;
        Vector3 smoothedPosition = Vector3.Lerp(transform.position, desiredPosition, smoothSpeed);
        transform.position = smoothedPosition;
    }
}
```

### 6.1 优化方案
- 使用LateUpdate避免抖动
- 添加边界限制防止越界

## 十二、性能优化方案
1. 物理优化：
   - 使用Physics2D.IgnoreLayerCollision优化不必要的碰撞
   - 减少Rigidbody2D的Fixed Update调用
   - 使用Box Collider 2D替代Polygon Collider

2. 渲染优化：
   - 使用Sprite Atlas打包所有精灵
   - 设置Sprite的Texture Type为Sprite (2D and UI)
   - 使用Culling Group优化远处物体

3. 内存优化：
   - 使用Object Pool管理频繁生成的物体
   - 定期清理无用的协程
   - 使用Addressables管理资源加载
